cmake_minimum_required(VERSION 2.6...3.18)

project(radar CXX)
set(CMAKE_WARN_UNUSED_CLI YES)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "" FORCE)
endif()
if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
endif()
option(BUILD_SHARED_LIBS "build shared libraries" ON)

find_package(OpenCV REQUIRED)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-pipe -mcmodel=medium -Wall -march=native)

add_subdirectory(yolo-trt)
add_subdirectory(bytetrack)
add_subdirectory(camera)
add_subdirectory(serial)
add_subdirectory(camera_calibration)
add_dependencies(calibration camera)
aux_source_directory(${CMAKE_CURRENT_SOURCE_DIR}/src ${PROJECT_NAME}_SOURCE)
add_executable(${PROJECT_NAME} ${${PROJECT_NAME}_SOURCE})
target_include_directories(${PROJECT_NAME} PRIVATE yolo-trt bytetrack camera serial ${OpenCV_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} yolo-trt bytetrack camera serial ${OpenCV_LIBS})

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_definitions(${PROJECT_NAME} PRIVATE -D_FORTIFY_SOURCE=2)
    target_compile_options(${PROJECT_NAME} PRIVATE -fstack-protector-strong -fstack-clash-protection)
elseif(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(${PROJECT_NAME} PRIVATE -D_GLIBCXX_ASSERTIONS -D_FORTIFY_SOURCE=1)
    target_compile_options(${PROJECT_NAME} PRIVATE -Og -fanalyzer -Wextra -Wpedantic)
endif()
